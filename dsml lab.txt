library(dplyr)

# Load dataset from web
url <- "https://stats.idre.ucla.edu/stat/data/binary.csv"
admission_data <- read_csv(url)

# Perform Logistic Regression (Correct formula)
model <- glm(admit ~ gre + gpa + rank,
             data = admission_data,
             family = binomial)

# Check the model fit
summary(model)

# Predict the probability of admission for new data
new_data <- data.frame(gre = 600, gpa = 3.5, rank = 3)
predicted_probability <- predict(model, newdata = new_data, type = "response")

print(paste("Predicted Probability of Admission:", predicted_probability))

library(dplyr)
library(ggplot2)

# Create dataset
df <- data.frame(
  state.abb = state.abb,
  population = runif(length(state.abb), 1000000, 50000000)
)

# Convert state abbreviation to factor (required for regression)
df$state.abb <- as.factor(df$state.abb)

# Regression model
model <- lm(population ~ state.abb, data = df)

# Summary
summary(model)

# Prediction
new_data <- data.frame(state.abb = factor(c("CA", "NY"), levels = levels(df$state.abb)))
predictions <- predict(model, newdata = new_data)
print(predictions)

# Plot
ggplot(df, aes(x = state.abb, y = population)) +
  geom_point() +
  geom_smooth(method = "lm")

import pandas as pd

# Load iris dataset
iris = pd.read_csv("iris.csv")

# ---------- SUBSET ----------
print("Subset: Rows where Species == 'setosa'")
subset_setosa = iris[iris["Species"] == "setosa"]
print(subset_setosa.head())

print("\nSubset: Sepal.Length > 5")
subset_length = iris[iris["SepalLength"] > 5]
print(subset_length.head())

# ---------- AGGREGATE ----------
print("\nAggregate: Mean of each Species")
agg_mean = iris.groupby("Species").mean()
print(agg_mean)

print("\nAggregate: Min, Max for each Species")
agg_min_max = iris.groupby("Species").agg(["min", "max"])
print(agg_min_max)
